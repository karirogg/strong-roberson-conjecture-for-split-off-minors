% Define some styles for cleaner code
\tikzset{
    vertex/.style={
        circle,
        draw,
        minimum size=4pt, % Adjust node size
        inner sep=0pt
    },
    label_node/.style={
        draw=none,
        inner sep=1pt,
        anchor=center,
        font=\small % Adjust font size if needed
    },
    c_label/.style={
        label_node,
        above,
        yshift=0.05cm % Shift "c_i" labels slightly further up
    }
}
